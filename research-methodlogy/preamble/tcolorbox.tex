	colback=tnBgHL, coltext=tnFg,
	colframe=tnGreen,
	size=small, boxrule=1pt,
	title=\textbf{\textcolor{tnGreen}{Contributions}},
	halign title=flush center,
	coltitle=tnGreen,
	breakable,
	drop shadow=tnBgDark!80!tnComment,
	attach boxed title to top left={xshift=1cm,yshift=-\tcboxedtitleheight/2,
		yshifttext=-\tcboxedtitleheight/2},
	minipage boxed title=1.5cm,
	boxed title style={%
		colback=tnBgDark, coltext=tnGreen,
		size=fbox, boxrule=1pt, boxsep=2pt,
		underlay={%
			\coordinate (dotA) at ($(interior.west) + (-0.5pt,0)$);
			\coordinate (dotB) at ($(interior.east) + (0.5pt,0)$);
			\begin{scope}
				\clip (interior.north west) rectangle ([xshift=3ex]interior.east);
				\filldraw [tnBgDark, blur shadow={shadow opacity=60, shadow yshift=-.75ex},
				rounded corners=2pt] (interior.north west) rectangle (interior.south east);
			\end{scope}
			\begin{scope}[tnGreen]
				\fill (dotA) circle (2pt);
				\fill (dotB) circle (2pt);
			\end{scope}
		},
	},
	#1,
}
